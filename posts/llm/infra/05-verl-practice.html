    async def generate(
        self,
        prompt_ids: torch.Tensor,
        sampling_params: dict[str, Any],
        request_id: str,
        image_data: Optional[list[Any]] = None,
    ) -> TokenOutput:
        """Generate sequence with token-in-token-out."""
        max_new_tokens = min(self.config.response_length, self.config.max_model_len - len(prompt_ids) - 1) 
        sampling_params["max_new_tokens"] = max_new_tokens
        return_logprob = sampling_params.pop("logprobs", False)

        request = GenerateReqInput(
            rid=request_id,
            input_ids=prompt_ids,
            sampling_params=sampling_params,
            return_logprob=return_logprob,
            image_data=image_data,
        )
        output = await self.tokenizer_manager.generate_request(request, None).__anext__()
        if return_logprob:
            output_token_logprobs = output["meta_info"]["output_token_logprobs"]
            log_probs, token_ids = zip(
                *[(log_prob, token_ids) for log_prob, token_ids, _ in output_token_logprobs], strict=True
            )
        else:
            token_ids = output["output_ids"]
            log_probs = None
        return TokenOutput(token_ids=token_ids, log_probs=log_probs)

batch_size相关 ​

简单总结 ​

batch_size 总结
  • data.train_batch_size
    • 训练batch大小，乘以rollout.n，得到real_train_batch_size
  • actor.ppo_mini_batch_size
    • 更新模型时，对real_train_batch划分为多个mini_batch
    • 1个real_train_batch 更新多次；1个mini_batch 更新1次
  • actor.ppo_micro_batch_size_per_gpu
    • 更新模型时，对mini_batch划分为多个micro_batch，依次计算，梯度累积
    • micro_batch_size=ppo_micro_batch_size_per_gpu
  • rollout.log_prob_micro_batch_size_per_gpu
    • 计算old_log_prob时，对real_train_batch划分为多个micro_batch，依次计算
  • ref.log_prob_micro_batch_size_per_gpu
    • 计算ref_log_prob时，对real_train_batch划分为多个micro_batch，依次计算

data.train_batch_size ​

data.train_batch_size

含义

  • train_batch_size：训练集batch大小，用于数据迭代加载训练
  • val_batch_size：同理， 不再赘述

关键点

  • 与rollout.n结合，构建出real_train_batch_size
    • real_train_batch_size = train_batch_size * rollout.n

actor.ppo_mini_batch_size ​

actor.ppo_mini_batch_size

前处理

  • MegatronActorRolloutRefWorker 构造函数里
  • 先自动放大n倍，再除以DP_world_size
    • 即：对actor.ppo_mini_batch_size 乘以 rollout.n
    • 再：除以get_data_parallel_world_size()
    • DP = N_GPUS // (MP * CP) = N_GPUS // (TP*PP*CP)

含义

  • update actor(或critic)时：对real_train_batch进行划分
  • 划分为多个mini_batch
    • 大小为ppo_mini_batch_size
    • 即real_train_batch_size -> 多个ppo_mini_batch_size
  • 每个mini_batch：更新1次模型
    • 通常与ppo_epochs结合使用，即rollout数据训练几次，重复使用几次。
    • 在1个real_train_batch内，分批次更新多次模型

Critic同理

  • 参数为critic.ppo_mini_batch_size

actor.ppo_micro_batch_size_per_gpu (梯度累积) ​

actor.ppo_micro_batch_size_per_gpu (梯度累积)

背景

  • 一次前向反向更新会OOM，因此拆分成多步。
  • 梯度累积：计算出梯度，但不清零，多步累积，最后一次性更新。

含义

  • update actor(或critic)时，对当前mini_batch进行划分
    • 划分为多个micro-batches
      • micro_batch_size = ppo_micro_batch_size_per_gpu
    • 本质就是梯度累积步数
  • real_train_batch_size -> ppo_mini_batch_size -> ppo_micro_batch_size_per_gpu

Critic同理

  • 参数为critic.ppo_mini_batch_size_per_gpu

以下3者有相似处

  • rollout.log_prob_micro_batch_size_per_gpu
    • 计算old_log_prob，划分real_train_batch
  • ref.log_prob_micro_batch_size_per_gpu
    • 计算ref_log_prob，划分real_train_batch
  • actor.ppo_micro_batch_size_per_gpu
    • 更新模型+梯度相关，划分mini-batch
ppo_micro_batch_size_per_gpu 注意事项
  • 这个值可能和RL学习稳定性有关系，小的数值+大的学习率，可能会导致训练崩塌。
  • 原因：单次梯度计算依赖的样本数量。
    • 如果太小，可能导致梯度估算的方差大，噪声大， 信噪比更低
    • 策略网络容易更新错误，被带偏了。
  • 需观测相关指标：pg_loss, kl_loss, entropy，advantages等等。

def forward_backward_batch(
        self,
        data: DataProto,
        forward_only=False,
        use_dynamic_bsz=False,
        micro_batch_size=None,
        max_token_len=None,
        mini_batch_size=None,
    ):
    # broadcast from last pp rank to all other pp ranks
    # ...
    mini_batch = data
    mini_batch.batch = mini_batch.batch.contiguous()
    # ...
    if use_dynamic_bsz:
        # pass
    else:
        micro_batches = mini_batch.batch.split(micro_batch_size)  
        seq_len = micro_batches[0]["input_ids"].shape[1] 
        total_seqlen = micro_batch_size * seq_len
    
    n_micro_batch = len(micro_batches) 
    forward_backward_func = get_forward_backward_func()
    # batch should be a list of batches inside micro-batches
    batch_generator = make_batch_generator(micro_batches, vpp_size=len(self.critic_module))  

    # TODO: we may use the new schedule instead
    # for flash-attn: (seq_len, batch_size, hidden_size) = (mbs*seq_len, 1, hidden_size)
    if mpu.get_pipeline_model_parallel_world_size() > 1:
        losses_reduced = forward_backward_func(
            forward_step_func=forward_step,
            data_iterator=batch_generator,  
            model=self.critic_module,
            num_microbatches=n_micro_batch,
            seq_length=total_seqlen,  # no use when input_shapes was set
            micro_batch_size=1,  # no use when input_shapes was set
            forward_only=forward_only,
        )
    else:
        losses_reduced = forward_backward_func( 
            forward_step_func=forward_step, 
            data_iterator=batch_generator,  
            model=self.critic_module, 
            num_microbatches=n_micro_batch, 
            seq_length=total_seqlen,  # in use for pp = 1 #
            micro_batch_size=1,  # in use for pp = 1 #
            forward_only=forward_only, 
        ) 
    # loss_reduces contains the stats returned from loss_func
    losses_reduced = {"output": losses_reduced} 
    if use_dynamic_bsz:
        losses_reduced["indices"] = indices
    return losses_reduced

    # batch should be a list of batches inside micro-batches
    batch_generator = make_batch_generator(micro_batches, vpp_size=len(self.critic_module))  

    # TODO: we may use the new schedule instead
    # for flash-attn: (seq_len, batch_size, hidden_size) = (mbs*seq_len, 1, hidden_size)
    if mpu.get_pipeline_model_parallel_world_size() > 1:
        losses_reduced = forward_backward_func(
            forward_step_func=forward_step,
            data_iterator=batch_generator,  
            model=self.critic_module,
            num_microbatches=n_micro_batch,
            seq_length=total_seqlen,  # no use when input_shapes was set
            micro_batch_size=1,  # no use when input_shapes was set
            forward_only=forward_only,
        )
    else:
        losses_reduced = forward_backward_func( 
            forward_step_func=forward_step, 
            data_iterator=batch_generator,  
            model=self.critic_module, 
            num_microbatches=n_micro_batch, 
            seq_length=total_seqlen,  # in use for pp = 1 #
            micro_batch_size=1,  # in use for pp = 1 #
            forward_only=forward_only, 
        ) 
    # loss_reduces contains the stats returned from loss_func
    losses_reduced = {"output": losses_reduced} 
    if use_dynamic_bsz:
        losses_reduced["indices"] = indices
    return losses_reduced

参数约束 ​

并行参数计算 ​

并行参数计算

实际训练bs

  • real_train_batch_size = data.train_batch_size * rollout.n

总GPU数量

  • n_gpus = n_gpus_per_node * nnodes

模型并行

  • mp = tp * pp
  • model_parallel_size = tensor_model_parallel_size * pipeline_model_parallel_size

数据并行

  • dp = n_gpus // (mp * cp)
  • megatron_dp = n_gpus // (model_parallel_size * context_parallel_size )

实际训练最小bs

  • mini_bs = dp * micro_bs_per_gpu
  • minimal_bsz = megatron_dp * actor.ppo_micro_batch_size_per_gpu

并行参数约束条件

实际train_bs被mini_bs整除

  • real_train_batch_size % minimal_bsz == 0

TP：注意力头数/GQA组数 整除

  • num_attention_heads % tp == 0
  • num_query_groups % tp == 0，非常重要!!

PP：总层数整除

  • num_layers % pp == 0

EP：专家数量整除

  • num_moe_experts % ep == 0

EP分组：专家数量整除

  • num_moe_experts % moe_router_num_groups == 0

Expert DP：专家数据并行度，DP 整除EP

  • Expert DP = DP / EP

TP/PP/CP参数：GPU数量整除

  • n_gpus % mp * cp == 0 ，非常重要!!
  • n_gpus % tp * pp * cp == 0
一些原理
  • TP：切分Attention/MLP
    • TP越大：单卡KV Cache压力小，支持Context越长
    • 缺点：受限于KV Heads数量；跨节点通信慢。
  • EP：切分Experts
    • EP越大：单价存单专家越少，静态显存占用低
    • 缺点：引入All-to-All通信，动态显存变大。长文本下，EP太大，Buffer可能撑爆显存。
  • 通信效率
    • 节点内：适合TP和EP通信
    • 节点间：适合DP通信，梯度同步，通信量小
    • 尽量让TP和EP都在一台机器内，解决。
